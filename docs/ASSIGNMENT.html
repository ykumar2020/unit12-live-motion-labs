Version 2 | November 9, 2026. Eight custom D3 experiences, native UI frameworks and free public APIs. This replaces the basic-chart edition. The supplied original assignment is unchanged.

1. Begin with Node.js

Install Node.js 22.12 or newer. Node runs JavaScript outside the browser. npm installs libraries; package-lock.json records dependency versions. Open a terminal in the extracted unit12-real-data-labs root, where package.json lives. Do not double-click index.html: these apps need their API server.

node --version
npm.cmd --version
npm.cmd ci
npm.cmd run dev

On macOS/Linux use npm instead of npm.cmd. Keep the terminal running and open http://127.0.0.1:5280. Eight servers run concurrently at ports 5281-5288. Stop with Ctrl+C or npm.cmd run stop. Windows launch/stop shortcuts are included. If a port is occupied, stop the conflicting process.

2. Honest live data, free APIs

Every load and refresh fetches the external provider. There is no saved-response fallback or local response cache. Weather and earthquakes refresh every two minutes while visible; other sources every five minutes. Fetch fresh data requests immediately. Providers can serve unchanged observations: fresh retrieval does not mean newly measured data.

USGS provides reported earthquakes; World Bank historical annual statistics; Open Library catalog metadata; NASA EONET curated natural events; Open-Meteo model forecasts. Annual playback is not a real-time population counter. Forecasts are not station observations. Animation changes layouts, never scientific values.

No key or paid subscription is required for these noncommercial classroom labs. Open-Meteo free access is noncommercial and quota-limited: https://open-meteo.com/en/pricing . Open Library requests low-volume access: https://openlibrary.org/developers/api . Do not load-test public services. Hidden tabs do not poll.

Inspect the data drawer for source URL, time, units, count and checksum. An explicit Download exports the response; it is never reused as fallback. Failure clears the scene and displays an error. EONET requests cap at 500 events and the radial scene displays up to 150; this is not a complete global hazard census.

3. JavaScript and D3 foundations

  • const declares a binding; let allows reassignment. Objects group fields and arrays hold records.
  • map transforms records; filter selects records. Missing null is not zero.
  • async functions return promises; await waits without blocking the UI. try/catch exposes errors.
  • SVG has selectable shapes. Canvas paints pixels efficiently for particle systems. D3 binds record IDs to marks.
  • Scales map data units to coordinates. Forces calculate layouts, not new measurements. Projections map geographic coordinates.
  • Stop timers and remove listeners on unmount. Reduced-motion starts static; Play is an explicit override.

7. Diagnose failures

  • npm not recognized: install Node and reopen the terminal. Windows execution-policy issue: use npm.cmd.
  • HTTP 502: read the provider error, inspect Network and retry later. Never invent a replacement record.
  • Identical refresh: compare request times/checksums. Provider publication or caching can leave values unchanged.
  • Build fails after dependency edits: restore the lockfile, npm ci, sync, build. Keep compatible dependency pins.
  • Drag events update geometry, not network requests. Respect free API quotas.
  • Submit all eight apps with custom interactions, attribution, units, responsive controls and honest failure behavior.

import { ref, onMounted, onBeforeUnmount, watch } from 'vue';
import { watchLive, recentYears } from './shared/sources.js';

const chart = ref(null),
  packet = ref(null),
  years = ref(25),

function load() {
  stop?.();
  packet.value = null;
  loading.value = true;
  error.value = '';
  chart.value?.replaceChildren();
  stop = watchLive(
    metric.value,
    (p) => {
      packet.value = p;
      loading.value = false;
    },
    (e) => {
      packet.value = null;
      loading.value = false;
      error.value = e.message;
    },
  );
}
function redraw() {
  if (!chart.value || !packet.value) return;
  const shown = recentYears(packet.value, years.value);

    kind: 'line',
    title:

      'six selected countries; the line may use a nonzero vertical origin. ' +
      'Forecasts and causal stories are not implied.',
  });
}
onMounted(load);
onBeforeUnmount(() => stop?.());

        filters: { first24Hours: firstDayOnly },
        limitation:
          'A model forecast is not an observed measurement or a climate trend. ' +
          'Times are displayed in America/New_York; the line breaks over ' +
          'missing hours.',
      });
  }, [packet, shown, firstDayOnly]);
  return (
    <>
      <div className="controls">
        <label>
          <input
            id="primary-control"
            aria-label="First 24 forecast hours"
            type="checkbox"
            checked={firstDayOnly}
            onChange={(e) => setFirstDayOnly(e.target.checked)}
          />{' '}
          Show first 24 forecast hours only
        </label>
      </div>
      {error ? (
        <p role="alert" className="error">
          {error}
        </p>
      ) : (
        <p role="status">
          {packet ? `${shown.length} hourly records selected` : 'Loading forecast…'}
        </p>
      )}
      <section className="chart-card">
        <h2>Temperature forecast (°C)</h2>
        <div ref={chart} className="chart" />
      </section>
    </>
  );
}

function liveAPI() {
  const install = (server) => {
    server.middlewares.use(async (req, res, next) => {
      const url = new URL(req.url, 'http://localhost');
      if (!url.pathname.startsWith('/api/data/')) return next();
      res.setHeader('Content-Type', 'application/json; charset=utf-8');
      res.setHeader('Cache-Control', 'no-store, no-cache, must-revalidate');
      const key = url.pathname.slice(10),
        source = Object.hasOwn(providers, key) ? providers[key] : null;
      if (!source) {
        res.writeHead(404);
        res.end(JSON.stringify({ error: 'Unknown live dataset' }));
        return;
      }
      if (req.method !== 'GET') {
        res.writeHead(405);
        res.end(JSON.stringify({ error: 'GET only' }));
        return;
      }
      if (url.searchParams.get('mode') === 'snapshot') {
        res.writeHead(410);
        res.end(
          JSON.stringify({
            error: 'Saved-data mode has been removed. This lab is live-only.',
          }),
        );
        return;
      }
      try {
        const response = await fetch(source.url, {
          cache: 'no-store',
          signal: AbortSignal.timeout(25000),
          headers: {
            'User-Agent': 'CPS5745-Live-D3-Labs/2.0',
            'Cache-Control': 'no-cache',
          },
        });
        if (!response.ok) throw Error(`${source.name} returned HTTP ${response.status}`);
        const rawText = await response.text(),
          raw = JSON.parse(rawText);
        res.end(
          JSON.stringify({
            raw,
            rawText,
            meta: {
              ...source,
              key,
              mode: 'live',
              fetchedAt: new Date().toISOString(),
              providerDate: response.headers.get('date'),
              providerAge: response.headers.get('age'),
              sha256: createHash('sha256').update(rawText).digest('hex'),
            },
          }),
        );
      } catch (error) {
        res.writeHead(502);
        res.end(
          JSON.stringify({
            error: `Live request failed: ${error.message}. No saved response is substituted.`,
          }),
        );
      }
    });
  };
  return {
    name: 'live-only-data',
    configureServer: install,
    configurePreviewServer: install,
  };
}

function domain(values) {
  let [a, b] = d3.extent(values);
  if (a === b && a === 0) return [0, 1];
  const pad = Math.max((b - a) * 0.07, a === b ? Math.abs(a) * 0.05 : 0, 0.00001);
  return [a - pad, b + pad];
}

export const numeric = (v) =>
  (typeof v === 'number' || (typeof v === 'string' && v.trim() !== '')) &&
  Number.isFinite(Number(v));
export function normalizeEarthquakes(raw) {
  if (!Array.isArray(raw?.features)) throw Error('USGS features[] missing');
  return raw.features
    .filter(
      (f) =>
        numeric(f.properties?.mag) && f.geometry?.coordinates?.length >= 3 && f.geometry.coordinates.slice(0, 3).every(numeric),
    )
    .map((f) => ({
      id: f.id,
      label: f.properties.place || 'Unknown location',
      x: +f.properties.mag,
      y: +f.geometry.coordinates[2],
      lon: +f.geometry.coordinates[0],
      lat: +f.geometry.coordinates[1],
      time: f.properties.time,
      url: f.properties.url,
    }));
}
export function normalizeWorldBank(raw) {
  if (!Array.isArray(raw?.[1])) throw Error('World Bank observations missing');
  if (+raw[0].pages > 1)
    throw Error('Response has more pages; refusing incomplete series');
  return raw[1]
    .filter((d) => numeric(d.value) && /^\d{4}$/.test(String(d.date)))
    .map((d) => ({
      label: `${d.country.value} · ${d.date}`,
      time: +d.date,
      value: +d.value,
      country: d.countryiso3code || d.country.id,
      countryName: d.country.value,
      year: +d.date,
    }))
    .sort((a, b) => a.time - b.time || a.country.localeCompare(b.country));
}
export function normalizeForecast(raw, variable) {
  const cities = ['New York', 'London', 'Mumbai', 'Singapore', 'São Paulo'];
  return (Array.isArray(raw) ? raw : [raw]).flatMap((r, cityIndex) => {
    const times = r.hourly?.time,
      values = r.hourly?.[variable];
    if (
      !Array.isArray(times) ||
      times.length !== values?.length ||
      r.hourly_units?.time !== 'unixtime'
    )
      throw Error('Invalid live hourly response');
    return times.flatMap((t, i) =>
      numeric(t) && numeric(values[i])
        ? [
            {
              label: new Date(t * 1000).toISOString(),
              time: t * 1000,
              value: +values[i],
              city: Array.isArray(raw) ? cities[cityIndex] : 'New York',
              lat: r.latitude,
              lon: r.longitude,
              wind: +r.hourly.wind_speed_10m?.[i],
              direction: +r.hourly.wind_direction_10m?.[i],
              humidity: +r.hourly.relative_humidity_2m?.[i],
            },
          ]
        : [],
    );
  });
}
export function normalizeBooks(raw) {
  if (!Array.isArray(raw.docs)) throw Error('Book results missing');
  return raw.docs
    .filter((d) => numeric(d.edition_count))
    .map((d) => ({
      id: d.key,
      label: d.title || 'Untitled',
      value: +d.edition_count,
      year: d.first_publish_year,
      authors: (d.author_key || []).map((id, i) => ({
        id,
        name: d.author_name?.[i] || id,
      })),
      subjects: (d.subject || []).slice(0, 5),
    }))
    .sort((a, b) => b.value - a.value);
}
export function normalizeEvents(raw, mode = 'all') {
  if (!Array.isArray(raw.events)) throw Error('NASA events missing');
  const counts = new Map();
  for (const event of raw.events) {
    let cats = [...new Set((event.categories || []).map((c) => c.title))];
    if (!cats.length) cats = ['Other'];
    if (mode === 'first') cats = cats.slice(0, 1);
    for (const label of cats) counts.set(label, (counts.get(label) || 0) + 1);
  }
  return [...counts]
    .map(([label, value]) => ({ label, value }))
    .sort((a, b) => b.value - a.value);
}
export function magnitudeBins(rows) {
  const map = new Map();
  for (const r of rows) {
    const k = Math.floor(r.x * 2) / 2;
    map.set(k, (map.get(k) || 0) + 1);
  }
  return [...map]
    .sort((a, b) => a[0] - b[0])
    .map(([k, value]) => ({
      label: `[${k.toFixed(1)}, ${(k + 0.5).toFixed(1)})`,
      value,
    }));
}
export function recentYears(packet, n) {
  const latest = Math.max(...packet.rows.map((d) => d.time));
  return packet.rows.filter((d) => d.time >= latest - n + 1);
}
export function firstHours(packet, hours) {
  const starts = new Map();
  for (const d of packet.rows)
    starts.set(d.city, Math.min(starts.get(d.city) ?? Infinity, d.time));
  return packet.rows.filter((d) => d.time < starts.get(d.city) + hours * 3600000);
}
export async function loadDataset(key, signal) {
  const response = await fetch(`/api/data/${key}`, { signal, cache: 'no-store' }),
    packet = await response.json();
  if (!response.ok) throw Error(packet.error || `HTTP ${response.status}`);
  if (packet.meta?.mode !== 'live') throw Error('Refusing non-live data');
  const raw = packet.raw;
  const rows =
    key === 'earthquakes'
      ? normalizeEarthquakes(raw)
      : ['population', 'growth'].includes(key)
        ? normalizeWorldBank(raw)
        : key === 'books'
          ? normalizeBooks(raw)
          : key === 'events'
            ? normalizeEvents(raw)
            : normalizeForecast(
                raw,
                key === 'temperature' ? 'temperature_2m' : 'precipitation',
              );
  return {
    ...packet,
    rows,
    meta: {
      ...packet.meta,
      validCount: rows.length,
      rawCount:
        key === 'earthquakes'
          ? raw.features.length
          : ['population', 'growth'].includes(key)
            ? raw[1].length
            : key === 'books'
              ? raw.docs.length
              : key === 'events'
                ? raw.events.length
                : rows.length,
      omitted: 0,
    },
  };
}
export function watchLive(key, onPacket, onError) {
  let controller,
    last,
    active = true,
    busy = false;
  async function refresh() {
    if (busy || document.hidden) return;
    busy = true;
    controller = new AbortController();
    document.dispatchEvent(
      new CustomEvent('lab:request', { detail: { key, state: 'fetching' } }),
    );
    try {
      const packet = await loadDataset(key, controller.signal);
      if (active) {
        last = packet;
        onPacket(packet);
        document.dispatchEvent(
          new CustomEvent('lab:request', {
            detail: { key, state: 'ready', at: packet.meta.fetchedAt },
          }),
        );
      }
    } catch (e) {
      if (active && e.name !== 'AbortError') {
        if (last) last.invalid = true;
        document.dispatchEvent(
          new CustomEvent('lab:request', {
            detail: { key, state: 'error', message: e.message },
          }),
        );
        onError(e);
      }
    } finally {
      busy = false;
    }
  }
  const interval = setInterval(
    refresh,
    ['population', 'growth', 'books', 'events'].includes(key) ? 300000 : 120000,
  );
  window.addEventListener('lab:refresh', refresh);
  refresh();
  return () => {
    active = false;
    controller?.abort();
    clearInterval(interval);
    window.removeEventListener('lab:refresh', refresh);
  };
}
